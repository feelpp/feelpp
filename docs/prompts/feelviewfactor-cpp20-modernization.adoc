== Overview

This document provides a comprehensive prompt for the phased modernization of the `feelpp/feel/feelviewfactor` module to C++20/23. The library computes view factors for radiative heat transfer using numerical integration and ray tracing methods.

=== Current State

[cols="1,2,1,1"]

| 449
| Major (public members, `using namespace`, poor encapsulation)

| `unobstructedplanarviewfactor.hpp`

=== Key Objectives

1. **C++20 Concepts**: Define `ViewFactorMesh` and `ViewFactorComputable` concepts
2. **Code Quality**: Fix bugs, encapsulate data, remove global `using namespace`
3. **Modern C++**: Use `std::format`, `std::span`, `[[nodiscard]]`, `constexpr`
4. **Parallelism**: Improve threading model with `std::jthread` (optional)
5. **C++23 Features**: Add `std::expected` for error handling (optional)
6. **Testing**: Ensure all changes are validated by compilation and test execution

=== Constraints

* C++20 minimum
* Must maintain backward API compatibility
* Tests must pass after each phase
* No changes to external dependencies (nanoflann, Eigen)

#ifndef FEELPP_FEELVIEWFACTOR_CONCEPTS_HPP
#define FEELPP_FEELVIEWFACTOR_CONCEPTS_HPP 1

#include <concepts>
#include <memory>
#include <string>
#include <vector>

namespace Feel
{

/**
 * @brief Concept for meshes compatible with view factor computation
 *
 * Requirements:
 * - Must have face_type for boundary iteration
 * - Must have nDim (2 or 3 only)
 * - Must support marker queries
 * - Must provide geometric mapping (gm())
 */
template<typename M>
concept ViewFactorMesh = requires(M m, std::string marker) {
    typename M::face_type;
    typename M::element_type;
    { M::nDim } -> std::convertible_to<int>;
    { m.hasMarker(marker) } -> std::convertible_to<bool>;
    { m.gm() };
    { m.numElements() } -> std::convertible_to<size_t>;
} && (M::nDim == 2 || M::nDim == 3);

/**
 * @brief Concept for view factor computation classes
 *
 * All view factor implementations must satisfy this concept.
 */
template<typename T>
concept ViewFactorComputable = requires(T t, bool elementwise) {
    typename T::value_type;
    typename T::mesh_t;
    { t.compute(elementwise) } -> std::same_as<void>;
    { t.viewFactors() };
    { t.areas() };
};

/**
 * @brief Concept for view factor base class compatibility
 */
template<typename T>
concept ViewFactorDerived = ViewFactorComputable<T> && requires(T t, unsigned int i, unsigned int j) {
    { t.devReciprocity(i, j) } -> std::convertible_to<typename T::value_type>;
    { t.maxDevReciprocity() } -> std::convertible_to<typename T::value_type>;
};

/**
 * @brief Concept for ray-traceable geometry
 *
 * Meshes that support BVH-based ray tracing.
 */
template<typename M>
concept RayTraceableMesh = ViewFactorMesh<M> && requires(M m) {
    // Must support trace mesh extraction
    typename trace_mesh_t<M>;
};

/**
 * @brief Concept for JSON configuration objects
 */
template<typename J>
concept ViewFactorConfig = requires(J j) {
    { j["viewfactor"]["markers"] };
    { j["viewfactor"]["quadrature_order"] };
};

} // namespace Feel

#endif /* FEELPP_FEELVIEWFACTOR_CONCEPTS_HPP */

#pragma once

#include <Eigen/Dense>
#include <feel/feelcore/feel.hpp>
#include <format>
#include <stdexcept>

namespace Feel::viewfactor::detail
{

/**
 * @brief Compute sum of areas of three triangles formed by point and triangle vertices (3D)
 */
[[nodiscard]] inline double elementArea(
    Eigen::VectorXd const& point,
    Eigen::VectorXd const& el_p1,
    Eigen::VectorXd const& el_p2,
    Eigen::VectorXd const& el_p3)
{
    if (point.size() != 3)
        throw std::logic_error(std::format(
            "elementArea (3D): expected 3D point, got dimension {}", point.size()));

    Eigen::Vector3d point_3d = point.head<3>();
    Eigen::Vector3d el_p1_3d = el_p1.head<3>();
    Eigen::Vector3d el_p2_3d = el_p2.head<3>();
    Eigen::Vector3d el_p3_3d = el_p3.head<3>();

    auto v1 = (point_3d - el_p1_3d).cross(point_3d - el_p2_3d);
    auto v2 = (point_3d - el_p2_3d).cross(point_3d - el_p3_3d);
    auto v3 = (point_3d - el_p3_3d).cross(point_3d - el_p1_3d);

    return v1.norm() / 2.0 + v2.norm() / 2.0 + v3.norm() / 2.0;
}

/**
 * @brief Compute sum of lengths of two segments (2D)
 */
[[nodiscard]] inline double elementArea(
    Eigen::VectorXd const& point,
    Eigen::VectorXd const& el_p1,
    Eigen::VectorXd const& el_p2)
{
    return (point - el_p1).norm() + (point - el_p2).norm();
}

/**
 * @brief Check if point lies on triangle surface (3D)
 */
[[nodiscard]] inline bool isOnSurface(
    Eigen::VectorXd const& point,
    Eigen::VectorXd const& el_p1,
    Eigen::VectorXd const& el_p2,
    Eigen::VectorXd const& el_p3,
    double tolerance = 1e-5)
{
    auto c = (el_p1 + el_p2 + el_p3) / 3.0;
    auto elem_area = elementArea(c, el_p1, el_p2, el_p3);
    auto area = elementArea(point, el_p1, el_p2, el_p3);
    return math::abs(area - elem_area) < tolerance;
}

/**
 * @brief Check if point lies on segment (2D)
 */
[[nodiscard]] inline bool isOnSurface(
    Eigen::VectorXd const& point,
    Eigen::VectorXd const& el_p1,
    Eigen::VectorXd const& el_p2,
    double tolerance = 1e-6)
{
    auto v1 = el_p2 - el_p1;
    auto v2 = point - el_p1;
    auto cross = v2[1] * v1[0] - v1[1] * v2[0];

    if (math::abs(cross) > tolerance)
        return false;

    if (math::abs(el_p2[0] - el_p1[0]) >= math::abs(el_p2[1] - el_p1[1]))
    {
        return (el_p2[0] - el_p1[0] > 0)
            ? (el_p1[0] <= point[0] && point[0] <= el_p2[0])
            : (el_p2[0] <= point[0] && point[0] <= el_p1[0]);
    }
    else
    {
        return (el_p2[1] - el_p1[1] > 0)
            ? (el_p1[1] <= point[1] && point[1] <= el_p2[1])
            : (el_p2[1] <= point[1] && point[1] <= el_p1[1]);
    }
}

} // namespace Feel::viewfactor::detail

#pragma once

#include <Eigen/Dense>
#include <cmath>
#include <format>
#include <numbers>
#include <random>
#include <span>
#include <stdexcept>

namespace Feel::viewfactor::detail
{

/**
 * @brief Generate random direction uniformly distributed on hemisphere (3D) or half-circle (2D)
 *
 * @param random_direction Output span to fill with direction components
 * @param gen1 First random number generator
 * @param gen2 Second random number generator
 * @param normal Surface normal vector
 */
inline void getRandomDirection(
    std::span<double> random_direction,
    std::mt19937& gen1,
    std::mt19937& gen2,
    Eigen::VectorXd const& normal)
{
    std::uniform_real_distribution<double> dist(0.0, 1.0);
    auto const size = static_cast<int>(random_direction.size());

    if (size == 3)
    {
        Eigen::Vector3d z_axis(0, 0, 1);
        double phi = 2.0 * std::numbers::pi * dist(gen1);
        double theta = std::asin(std::sqrt(dist(gen2)));

        random_direction[0] = std::sin(theta) * std::cos(phi);
        random_direction[1] = std::sin(theta) * std::sin(phi);
        random_direction[2] = std::cos(theta);

        // Rotate if normal is not aligned with z-axis
        if (std::abs(normal.dot(z_axis)) < 1.0)
        {
            Eigen::Vector3d direction(random_direction[0], random_direction[1], random_direction[2]);
            Eigen::Vector3d crossProd1 = z_axis.cross(normal.head<3>());
            Eigen::Vector3d crossProd2 = crossProd1.cross(z_axis);
            Eigen::Vector3d crossProd3 = crossProd1.cross(normal.head<3>());

            Eigen::Matrix3d matrix1, matrix2;
            matrix1.col(0) = z_axis;
            matrix1.col(1) = crossProd2;
            matrix1.col(2) = crossProd1;

            matrix2.row(0) = normal.head<3>();
            matrix2.row(1) = crossProd3;
            matrix2.row(2) = crossProd1;

            direction = matrix1 * (matrix2 * direction);

            random_direction[0] = direction[0];
            random_direction[1] = direction[1];
            random_direction[2] = direction[2];
        }
    }
    else if (size == 2)
    {
        double phi = std::numbers::pi * dist(gen1);
        random_direction[0] = std::cos(phi);
        random_direction[1] = std::sin(phi);
    }
    else
    {
        throw std::logic_error(std::format(
            "Wrong dimension {} for the random direction (expected 2 or 3)", size));
    }
}

} // namespace Feel::viewfactor::detail
